------Creat Seurat Object
Control <- Read10X(data.dir ="H:/Control/filtered_feature_bc_matrix/")
Control <- CreateSeuratObject(counts = Control, project = "Control")
EGFR <- Read10X(data.dir ="H:/EGFR/filtered_feature_bc_matrix/")
EGFR <- CreateSeuratObject(counts = EGFR, project = "EGFR")
------Filtering
Control[["percent.mt"]] <- PercentageFeatureSet(Control, pattern = "^mt-")
EGFR[["percent.mt"]] <- PercentageFeatureSet(EGFR, pattern = "^mt-")
Control <- subset(Control, subset = nFeature_RNA > 200 & nFeature_RNA < 6000 & percent.mt < 10)
EGFR <- subset(EGFR, subset = nFeature_RNA > 200 & nFeature_RNA < 6000 & percent.mt < 10)
------Preprocessing
Control$Sample <- "Control"
Control<- NormalizeData(Control, normalization.method = "LogNormalize", scale.factor = 10000)
Control<- FindVariableFeatures(Control, selection.method = "vst", nfeatures = 2000)
EGFR$Sample <- "EGFR"
EGFR<- NormalizeData(EGFR, normalization.method = "LogNormalize", scale.factor = 10000)
EGFR<- FindVariableFeatures(EGFR, selection.method = "vst", nfeatures = 2000)
immune.anchors <- FindIntegrationAnchors(object.list = list(Control,EGFR), dims = 1:20)
remove(Control,EGFR)
immune.combined <- IntegrateData(anchorset = immune.anchors, dims = 1:20)
--------------Batch Correction after Dimensionality Reduction
immune.combined <- ScaleData(immune.combined)
immune.combined <- RunPCA(immune.combined, features = VariableFeatures(object = immune.combined))
immune.combined  <- RunHarmony(combined, group.by.vars = "Sample", plot_convergence = TRUE)
-------------- Dimensionality Reduction ====================
immune.combined <- FindNeighbors(immune.combined, dims = 1:20)
DefaultAssay(immune.combined) <- "integrated"
immune.combined <- FindClusters(immune.combined, resolution = 0.3)
immune.combined <- RunUMAP(immune.combined, dims = 1:20)
DefaultAssay(immune.combined) <- "RNA"
DimPlot(immune.combined, reduction = "umap", label = T)
-------------- Find the marker genes
Clusters <- FindAllMarkers(immune.combined, only.pos = TRUE, min.pct = 0.25, logfc.threshold = 0.25)
